## 🔧 技术要点

### 1. API 服务层使用

```javascript
// 优化前
const request = time.Promisify(wx.request);
const res = await request({
  url: app.host + '/singer',
  data: { id, page, size }
});

// 优化后
const API = require('../../services/api');
const res = await API.singer.getSingerSongs({
  id, page, size
});
```

### 2. 数据缓存使用

```javascript
const cache = require('../../utils/cache');

// 获取缓存
const cached = await cache.get('recommen_1');
if (cached) {
  // 使用缓存数据
  return;
}

// 获取新数据并缓存
const res = await API.playlist.getRecommendInfo({ id, page });
if (res.success) {
  await cache.set('recommen_1', res.data, 3600000); // 缓存1小时
}
```

### 3. 图片懒加载

```xml
<!-- 优化前 -->
<image src="{{item.img}}" />

<!-- 优化后 -->
<image src="{{item.img}}" lazy-load="{{true}}" />
```

### 4. 通用组件使用

```xml
<!-- Button 组件 -->
<custom-button 
  text="确认" 
  type="primary" 
  size="medium"
  bindtap="onConfirm"
/>

<!-- Card 组件 -->
<custom-card 
  title="歌单名称"
  description="歌单描述"
  image="{{item.img}}"
  clickable="{{true}}"
  bindtap="onCardTap"
/>
```
