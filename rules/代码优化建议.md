# wx-music 代码优化建议

## 📋 概述

本文档基于对 `wx-music` 项目的代码分析，结合 `wx_music_serve` 后端服务的优化经验，提出代码层面的优化建议。

## 🔍 发现的主要问题

### 1. API 请求管理混乱

**问题描述**：
- 项目中存在大量直接使用 `wx.request` 的代码（38处）
- 没有统一的请求封装和错误处理
- API 地址硬编码在多个文件中
- 缺少请求拦截器和响应拦截器

**影响**：
- 代码重复度高
- 错误处理不统一
- 难以维护和调试
- 无法统一处理 token、loading 等

**优化建议**：

#### 1.1 创建统一的请求工具类

创建 `utils/request.js`：

```javascript
const app = getApp();
const apiConfig = require('../pages/api/index');

class Request {
  constructor() {
    this.baseURL = apiConfig.host;
    this.timeout = 10000;
    this.loadingCount = 0;
  }

  // 显示加载
  showLoading(title = '加载中') {
    this.loadingCount++;
    if (this.loadingCount === 1) {
      wx.showLoading({ title, mask: true });
    }
  }

  // 隐藏加载
  hideLoading() {
    this.loadingCount--;
    if (this.loadingCount <= 0) {
      this.loadingCount = 0;
      wx.hideLoading();
    }
  }

  // 统一请求方法
  async request(options) {
    const {
      url,
      method = 'GET',
      data = {},
      header = {},
      showLoading = true,
      showError = true
    } = options;

    // 显示加载
    if (showLoading) {
      this.showLoading();
    }

    try {
      // 获取用户 token（如果有）
      const openId = await this.getOpenId();
      if (openId) {
        header['X-User-Id'] = openId;
      }

      // 发起请求
      const res = await new Promise((resolve, reject) => {
        wx.request({
          url: url.startsWith('http') ? url : `${this.baseURL}${url}`,
          method,
          data,
          header: {
            'Content-Type': 'application/json',
            ...header
          },
          timeout: this.timeout,
          success: resolve,
          fail: reject
        });
      });

      // 隐藏加载
      if (showLoading) {
        this.hideLoading();
      }

      // 统一处理响应
      if (res.statusCode === 200) {
        const { code, data, message } = res.data;
        
        if (code === 200) {
          return { success: true, data, message };
        } else {
          // 业务错误
          if (showError) {
            wx.showToast({
              title: message || '请求失败',
              icon: 'none'
            });
          }
          return { success: false, message, code };
        }
      } else {
        throw new Error(`HTTP ${res.statusCode}`);
      }
    } catch (error) {
      // 隐藏加载
      if (showLoading) {
        this.hideLoading();
      }

      // 错误处理
      const errorMsg = this.getErrorMessage(error);
      if (showError) {
        wx.showToast({
          title: errorMsg,
          icon: 'none'
        });
      }
      return { success: false, message: errorMsg };
    }
  }

  // 获取 OpenID
  async getOpenId() {
    try {
      const res = await wx.getStorage({ key: 'openId', encrypt: true });
      return res.data;
    } catch {
      return null;
    }
  }

  // 错误信息处理
  getErrorMessage(error) {
    if (error.errMsg) {
      if (error.errMsg.includes('timeout')) {
        return '请求超时，请检查网络';
      }
      if (error.errMsg.includes('fail')) {
        return '网络连接失败';
      }
    }
    return error.message || '请求失败，请稍后重试';
  }

  // GET 请求
  get(url, data, options = {}) {
    return this.request({ ...options, url, method: 'GET', data });
  }

  // POST 请求
  post(url, data, options = {}) {
    return this.request({ ...options, url, method: 'POST', data });
  }

  // PUT 请求
  put(url, data, options = {}) {
    return this.request({ ...options, url, method: 'PUT', data });
  }

  // DELETE 请求
  delete(url, data, options = {}) {
    return this.request({ ...options, url, method: 'DELETE', data });
  }

  // 文件上传
  async upload(options) {
    const {
      url,
      filePath,
      name = 'file',
      formData = {},
      showLoading = true,
      showError = true
    } = options;

    if (showLoading) {
      this.showLoading('上传中');
    }

    try {
      const openId = await this.getOpenId();
      if (openId) {
        formData.userId = openId;
      }

      const res = await new Promise((resolve, reject) => {
        wx.uploadFile({
          url: url.startsWith('http') ? url : `${this.baseURL}${url}`,
          filePath,
          name,
          formData,
          success: resolve,
          fail: reject
        });
      });

      if (showLoading) {
        this.hideLoading();
      }

      const data = JSON.parse(res.data);
      if (data.code === 200) {
        return { success: true, data: data.data, message: data.message };
      } else {
        if (showError) {
          wx.showToast({
            title: data.message || '上传失败',
            icon: 'none'
          });
        }
        return { success: false, message: data.message };
      }
    } catch (error) {
      if (showLoading) {
        this.hideLoading();
      }
      const errorMsg = this.getErrorMessage(error);
      if (showError) {
        wx.showToast({
          title: errorMsg,
          icon: 'none'
        });
      }
      return { success: false, message: errorMsg };
    }
  }
}

// 导出单例
const request = new Request();
module.exports = request;
```

### 2. 错误处理不统一

**问题描述**：
- 每个页面都有独立的错误处理逻辑
- 错误提示方式不一致
- 缺少全局错误处理机制

**优化建议**：

#### 2.1 创建错误处理工具

创建 `utils/errorHandler.js`：

```javascript
class ErrorHandler {
  // 处理 API 错误
  handleApiError(error, customMessage) {
    let message = customMessage;
    
    if (!message) {
      if (error.code) {
        // 业务错误码
        message = this.getBusinessErrorMessage(error.code);
      } else if (error.errMsg) {
        // 网络错误
        message = this.getNetworkErrorMessage(error.errMsg);
      } else {
        message = '操作失败，请稍后重试';
      }
    }
    
    wx.showToast({
      title: message,
      icon: 'none',
      duration: 2000
    });
    
    // 记录错误日志（开发环境）
    if (process.env.NODE_ENV === 'development') {
      console.error('API Error:', error);
    }
  }

  // 业务错误码映射
  getBusinessErrorMessage(code) {
    const errorMap = {
      400: '请求参数错误',
      401: '未授权，请重新登录',
      403: '没有权限',
      404: '资源不存在',
      500: '服务器错误',
      503: '服务暂不可用'
    };
    return errorMap[code] || '操作失败';
  }

  // 网络错误处理
  getNetworkErrorMessage(errMsg) {
    if (errMsg.includes('timeout')) {
      return '请求超时，请检查网络';
    }
    if (errMsg.includes('fail')) {
      return '网络连接失败';
    }
    return '网络错误，请稍后重试';
  }
}

module.exports = new ErrorHandler();
```

### 4. 状态管理混乱

**问题描述**：
- 使用 `app.data` 存储全局状态
- 状态更新不统一
- 缺少状态管理机制

**优化建议**：

#### 4.1 创建状态管理工具

创建 `utils/store.js`：

```javascript
class Store {
  constructor() {
    this.state = {
      userInfo: null,
      currentSong: null,
      playList: [],
      playStatus: 'pause' // pause, playing, loading
    };
    this.listeners = [];
  }

  // 获取状态
  getState(key) {
    return key ? this.state[key] : this.state;
  }

  // 设置状态
  setState(updates) {
    const oldState = { ...this.state };
    this.state = { ...this.state, ...updates };
    this.notifyListeners(oldState, this.state);
  }

  // 订阅状态变化
  subscribe(listener) {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(l => l !== listener);
    };
  }

  // 通知监听器
  notifyListeners(oldState, newState) {
    this.listeners.forEach(listener => {
      listener(newState, oldState);
    });
  }
}

module.exports = new Store();
```

### 5. 代码质量问题

**问题描述**：
- 存在大量注释掉的代码
- 变量命名不规范（如 `paegSize` 应为 `pageSize`）
- 缺少代码注释
- 魔法数字和字符串

**优化建议**：

#### 5.1 清理注释代码
- 删除所有注释掉的代码
- 使用 Git 版本控制管理历史代码

#### 5.2 统一命名规范
- 使用驼峰命名法
- 变量名要有意义
- 常量使用大写字母和下划线

#### 5.3 提取常量
